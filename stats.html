Sambungan Google Sheets

Dengan menyambungkan Google Sheets, semua markah dan nama pelajar yang menduduki ujian di mana-mana peranti akan terus masuk ke spreadsheet Google Drive anda secara automatik dan percuma.

Panduan Setup 3 Minit:

  1. Buka Google Drive dan buat satu Google Sheets baru.
  2. Di Google Sheets, klik menu Extensions > Apps Script.
  3. Padam semua kod sedia ada, dan tampal kod Apps Script di bawah:
function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var rows = sheet.getDataRange().getValues();
  if (rows.length <= 1) {
    return ContentService.createTextOutput(JSON.stringify([]))
      .setMimeType(ContentService.MimeType.JSON);
  }
  var results = [];
  for (var i = 1; i < rows.length; i++) {
    var r = rows[i];
    if (!r[0] && !r[2]) continue;
    results.push({
      id: r[0] || ('AD-' + i),
      timestamp: r[1] ? String(r[1]) : '-',
      name: r[2] || 'Calon (Tanpa Nama)',
      group: r[3] || '-',
      score: Number(r[4]) || 0,
      total: Number(r[5]) || 10,
      percentage: Number(r[6]) || 0,
      status: r[7] || (Number(r[6]) >= 75 ? 'LULUS' : 'GAGAL')
    });
  }
  return ContentService.createTextOutput(JSON.stringify(results))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(["ID Rekod", "Tarikh & Masa", "Nama Calon", "Program / Kelas", "Markah", "Jumlah Soalan", "Peratus (%)", "Status"]);
    }
    var data = JSON.parse(e.postData.contents);
    sheet.appendRow([
      data.id || ('AD-' + Date.now().toString().slice(-6)),
      data.timestamp || new Date().toLocaleString(),
      data.name || 'Calon (Tanpa Nama)',
      data.group || '-',
      data.score !== undefined ? data.score : 0,
      data.total !== undefined ? data.total : 10,
      data.percentage !== undefined ? data.percentage : 0,
      data.status || (data.percentage >= 75 ? 'LULUS' : 'GAGAL')
    ]);
    return ContentService.createTextOutput(JSON.stringify({ status: "success" }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
